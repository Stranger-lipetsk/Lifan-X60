--- setup_btmac.sh ---
#!/system/bin/sh

BT_ADDR_STR="bt_addr"
CONFIG_FILE="/mnt/para_fw/config.ini"

while read -r LINE; do
	LINE_ARRY=(${LINE//=/ })
	if [ ${LINE_ARRY[0]} == $BT_ADDR_STR ]; then
		#MAC_STR=${LINE_ARRY[1]}
		MAC_STR=${LINE_ARRY[1]:0:2}":"${LINE_ARRY[1]:2:2}":"${LINE_ARRY[1]:4:2}":"${LINE_ARRY[1]:6:2}":"${LINE_ARRY[1]:8:2}":"${LINE_ARRY[1]:10:2}
		mkdir -p /data/bluesoleil/
		chmod 777 /data/bluesoleil/
		echo $MAC_STR > /data/bluesoleil/BD_ADDR.INI
		chmod 666 /data/bluesoleil/BD_ADDR.INI
		cp /system/etc/bluesoleil/TL_BASE.INI /data/bluesoleil/TL.INI
		echo "bd_addr="$MAC_STR >> /data/bluesoleil/TL.INI
		chmod 666 /data/bluesoleil/TL.INI
		setprop ro.bt.bdaddr_path $MAC_STR
		sync
		sync
		setprop rw.set_btmac_finished 1
		exit 0
	fi	
done<$CONFIG_FILE

setprop rw.set_btmac_finished 1
exit 0
